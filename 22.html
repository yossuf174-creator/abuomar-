import 'package:flutter/material.dart'; class TextAndLinksToolScreen extends StatefulWidget { const TextAndLinksToolScreen({Key? key}) : super(key: key); @override _TextAndLinksToolScreenState createState() => _TextAndLinksToolScreenState(); } class _TextAndLinksToolScreenState extends State { final TextEditingController _inputController = TextEditingController(); List _extractedLinks = []; String _textStats = ''; // دالة لتحليل النص واستخراج الروابط والمعلومات الأساسية void _processText() { String text = _inputController.text; // تعبير منتظم (Regex) لاستخراج الروابط من النص RegExp regExp = RegExp( r'((https?:\/\/)?(www\.)?[a-zA-Z0-9@:%._\+~#=]{1,256}\.[a-zA-Z0-9()]{1,6}\b([-a-zA-Z0-9()@:%_\+.~#?&//=]*))', caseSensitive: false, ); Iterable matches = regExp.allMatches(text); setState(() { _extractedLinks = matches.map((m) => m.group(0)!).toList(); int charCount = text.length; int wordCount = text.trim().isEmpty ? 0 : text.trim().split(RegExp(r'\s+')).length; int lineCount = text.isEmpty ? 0 : text.split('\n').length; _textStats = 'عدد الكلمات: $wordCount | عدد الحروف: $charCount | عدد الأسطر: $lineCount'; }); } @override Widget build(BuildContext context) { return Scaffold( appBar: AppBar( title: const Text('أداة النصوص والروابط'), centerTitle: true, ), body: Padding( padding: const EdgeInsets.all(16.0), child: Column( crossAxisAlignment: CrossAxisAlignment.stretch, children: [ // حقل إدخال النص TextField( controller: _inputController, maxLines: 5, decoration: const InputDecoration( labelText: 'الصق النص أو الروابط هنا', border: OutlineInputBorder(), hintText: 'اكتب أو الصق النصوص هنا لمعالجتها...', ), ), const SizedBox(height: 12), // زر المعالجة ElevatedButton.icon( onPressed: _processText, icon: const Icon(Icons.analytics), label: const Text('تحليل واستخراج الروابط والمعلومات'), style: ElevatedButton.styleFrom( padding: const EdgeInsets.symmetric(vertical: 12), ), ), const SizedBox(height: 16), // عرض إحصائيات النص الأساسية if (_textStats.isNotEmpty) Card( color: Colors.blue.shade900, child: Padding( padding: const EdgeInsets.all(12.0), child: Text( _textStats, style: const TextStyle(fontSize: 14, fontWeight: FontWeight.bold), textAlign: TextAlign.center, ), ), ), const SizedBox(height: 10), // عرض الروابط المستخرجة const Text( 'الروابط المكتشفة:', style: TextStyle(fontSize: 16, fontWeight: FontWeight.bold), ), const SizedBox(height: 8), Expanded( child: _extractedLinks.isEmpty const Center(child: Text('لا توجد روابط مستخرجة حالياً')) : ListView.builder( itemCount: _extractedLinks.length, itemBuilder: (context, index) { return Card( child: ListTile( leading: const Icon(Icons.link, color: Colors.blue), title: Text( _extractedLinks[index], style: const TextStyle(color: Colors.blueAccent), ), ), ); }, ), ), ], ), ), ); } }
